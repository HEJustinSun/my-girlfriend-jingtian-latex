% Latin fonts carry +pwid: Source Han CN draws U+2019 full-width by CJK
% convention, which wrecks every English apostrophe in the dialogue. pwid
% selects the proportional form. The CJK fonts below deliberately omit it,
% so Chinese quotation marks keep their full-width shape.
% Ligatures=TeX is likewise dropped: the source already carries real
% Unicode dashes and quotes, so the TeX ligature set buys nothing.
\defaultfontfeatures{Ligatures=NoCommon}
\setmainfont[
  Path=../../../fonts/, RawFeature={+pwid},
  BoldFont=SourceHanSerifCN-Regular.otf, BoldFeatures={FakeBold=2.2},
  ItalicFont=SourceHanSerifCN-Regular.otf, ItalicFeatures={FakeSlant=0.15}
]{SourceHanSerifCN-Regular.otf}
\setsansfont[
  Path=../../../fonts/, RawFeature={+pwid},
  BoldFont=SourceHanSansCN-Medium.otf, BoldFeatures={FakeBold=2.0},
  ItalicFont=SourceHanSansCN-Medium.otf, ItalicFeatures={FakeSlant=0.15}
]{SourceHanSansCN-Medium.otf}
\setCJKmainfont[Path=../../../fonts/,AutoFakeBold=2.2,AutoFakeSlant=0.15]{SourceHanSerifCN-Regular.otf}
\setCJKsansfont[Path=../../../fonts/,AutoFakeBold=2.0,AutoFakeSlant=0.15]{SourceHanSansCN-Medium.otf}
\setCJKmonofont[Path=../../../fonts/]{SourceHanSansCN-Medium.otf}
\xeCJKsetup{PunctStyle=plain}
% U+2014 does double duty here: 257 of its 277 uses are English em dashes
% in sluglines and episode titles, where xeCJK's CJK class swallows the
% following space ("CLINIC —DAY"). Class it as Latin, and let \CNdash
% re-enter the CJK class for the 10 Chinese 破折号.
\xeCJKDeclareCharClass{Default}{"2014}
\newcommand{\CNdash}{\begingroup\xeCJKDeclareCharClass{CJK}{"2014}——\endgroup}

% Reserve vertical space so a heading is never orphaned at a page bottom.
% Stands in for the needspace package, which this TeX Live cannot install.
\newcommand{\KeepTogether}[1]{%
  \begingroup
  \setlength{\dimen@}{#1}%
  \vskip 0pt plus \dimen@
  \penalty -100
  \vskip 0pt plus -\dimen@
  \vskip 0.0001pt
  \endgroup
}
